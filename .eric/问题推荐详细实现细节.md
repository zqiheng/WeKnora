```go
// SuggestionItem 是一个稳定、可归因、渲染给终端用户的问题。
type SuggestionItem struct {
	ID               string   `json:"id"`
	Text             string   `json:"text"`
	Category         string   `json:"category,omitempty"`
	Source           string   `json:"source"`
	KnowledgeBaseIDs []string `json:"knowledge_base_ids,omitempty"`
}

// SuggestionItems 是 []SuggestionItem 的别名，实现 driver.Valuer/sql.Scanner。
type SuggestionItems []SuggestionItem

func (s SuggestionItems) Value() (driver.Value, error) {
	if s == nil {
		s = SuggestionItems{}
	}
	return json.Marshal(s)
}

func (s *SuggestionItems) Scan(value interface{}) error {
	if value == nil {
		*s = SuggestionItems{}
		return nil
	}
	var b []byte
	switch v := value.(type) {
	case []byte:
		b = v
	case string:
		b = []byte(v)
	default:
		*s = SuggestionItems{}
		return nil
	}
	return json.Unmarshal(b, s)
}
```

关键点：`Value()` 把 `nil` 归一化为 `[]`（而非 `null`）；`Scan()` 兼容 `[]byte`/`string` 两种驱动返回值。

- `questions` 列：Postgres `JSONB DEFAULT '[]'::jsonb`，SQLite `TEXT DEFAULT '[]'`，MySQL `JSON NOT NULL`。

- `latency_ms`：SQLite `INTEGER`（无 BIGINT）。
- MySQL `updated_at` 用 `ON UPDATE CURRENT_TIMESTAMP`。

#### `AcquireGeneration`（核心：DB 行 + 过期租约做分布式互斥）

```go
func (r *messageSuggestionRepository) AcquireGeneration(
	ctx context.Context,
	candidate *types.MessageSuggestionSet,
	regenerate bool,
) (*types.MessageSuggestionSet, bool, error) {
	now := time.Now()
	leaseUntil := now.Add(3 * time.Minute)
	candidate.Status = types.SuggestionStatusGenerating
	candidate.LeaseUntil = &leaseUntil
	candidate.Questions = types.SuggestionItems{}

	// 第一步：靠唯一索引冲突 + ON CONFLICT DO NOTHING 原子抢占
	result := r.db.WithContext(ctx).
		Clauses(clause.OnConflict{DoNothing: true}).
		Create(candidate)
	if result.Error != nil {
		return nil, false, result.Error
	}
	if result.RowsAffected == 1 {
		return candidate, true, nil // 新建成功 = 抢到租约
	}

	// 第二步：已存在，读现有行判断
	existing, err := r.GetByCacheKey(ctx, candidate.TenantID, candidate.AssistantMessageID,
		candidate.Placement, candidate.ConfigHash, candidate.Locale)
	if err != nil {
		return nil, false, err
	}
	if existing.Status == types.SuggestionStatusReady && !regenerate {
		return existing, false, nil // ready 命中 → 直接复用
	}
	if existing.Status == types.SuggestionStatusSuppressed && !regenerate {
		return existing, false, nil // suppressed 命中 → 直接复用
	}
	if existing.Status == types.SuggestionStatusGenerating &&
		existing.LeaseUntil != nil && existing.LeaseUntil.After(now) {
		return existing, false, nil // 别人正在生成 → 让出
	}

	// 第三步：二次抢占（租约已过期 / 非 generating / regenerate）
	query := r.db.WithContext(ctx).Model(&types.MessageSuggestionSet{}).
		Where("id = ?", existing.ID).
		Where("status <> ? OR lease_until IS NULL OR lease_until < ?",
			types.SuggestionStatusGenerating, now)
	if existing.Status == types.SuggestionStatusReady && regenerate {
		query = r.db.WithContext(ctx).Model(&types.MessageSuggestionSet{}).
			Where("id = ? AND status = ?", existing.ID, types.SuggestionStatusReady)
	}
	updates := map[string]interface{}{
		"status":             types.SuggestionStatusGenerating,
		"lease_until":        leaseUntil,
		"suppression_reason": "",
		"questions":          types.SuggestionItems{},
		"error_code":         "",
		"generated_at":       nil,
		"updated_at":         now,
	}
	result = query.Updates(updates)
	if result.Error != nil {
		return nil, false, result.Error
	}
	if result.RowsAffected == 0 {
		current, getErr := r.GetByCacheKey(ctx, candidate.TenantID, candidate.AssistantMessageID,
			candidate.Placement, candidate.ConfigHash, candidate.Locale)
		return current, false, getErr
	}
	existing.Status = types.SuggestionStatusGenerating
	existing.LeaseUntil = &leaseUntil
	existing.Questions = types.SuggestionItems{}
	return existing, true, nil
}
```

**等效 SQL**：

```sql
-- 第一步：INSERT ... ON CONFLICT DO NOTHING
-- （MySQL 转 INSERT IGNORE，SQLite 转 OR IGNORE）

-- 第三步：抢占 UPDATE（乐观条件）
UPDATE message_suggestion_sets
SET status = 'generating', lease_until = ?, suppression_reason = '',
    questions = ?, error_code = '', generated_at = NULL, updated_at = ?
WHERE id = ? AND (status <> 'generating' OR lease_until IS NULL OR lease_until < ?);

-- regenerate && status='ready' 时的替换 WHERE：
UPDATE message_suggestion_sets
SET status = 'generating', lease_until = ?, ...
WHERE id = ? AND status = 'ready';
```

**返回值语义**：`(set, acquired, error)`。`acquired == true` 表示调用方获得生成权应执行生成；`false` 表示已有他人持有（返回最新 `existing`，调用方直接复用）。

### 1.6 可移植要点（持久层）

1. **「DB 行 + 过期租约」做分布式互斥**：先 `INSERT ... ON CONFLICT DO NOTHING` 靠唯一索引抢占，`RowsAffected==1` 即成功；冲突后读现有行，`generating` 且租约未过期则让出，否则用带乐观 WHERE 的 `UPDATE` 二次抢占。无锁、可崩溃自动恢复。
2. **`config_hash` 参与缓存键**：同一助手消息 + 不同 agent 配置 → 哈希不同 → 生成全新 set，配置变了天然失效缓存。
3. **事件表引用 ID 不复制文本**：`MessageSuggestionEvent.QuestionID` 只存稳定 ID，文案改动不影响历史事件。
4. **`SuggestionItems` 自定义 Valuer/Scanner**：Go 侧统一、DB 侧按方言，nil 归一为 `[]`。
5. **JSON 字段按 `db.Name()` 三方言显式分支**：移植到单一数据库时只保留对应方言。

### 2.6 Validate（校验，`:371-419`）

边界汇总：
- `Starters.Count`：1~8；`FollowUps.Count`：1~5；`FollowUps.MaxContextTurns`：1~5。
- `Starters.Mode` 白名单：`curated`/`knowledge`/`hybrid`；`FollowUps.Mode` 白名单：`generated`/`knowledge`/`hybrid`。

### 3.3 能力过滤 `DeriveKBFilterForAgent`（`internal/agent/tools/capabilities.go`）

```go
var quickAnswerKBFilter = KBFilter{AnyOf: []KBCapability{CapVector, CapKeyword}}

func DeriveKBFilterForAgent(agentMode string, allowedTools []string) KBFilter {
	seen := make(map[KBCapability]struct{})
	if agentMode == "quick-answer" { // 隐式约束：quick-answer 强制 vector|keyword
		for _, c := range quickAnswerKBFilter.AnyOf {
			seen[c] = struct{}{}
		}
	}
	for _, c := range DeriveKBFilterFromTools(allowedTools).AnyOf { // 工具派生约束（并集）
		seen[c] = struct{}{}
	}
	// ... 转成 KBFilter{AnyOf: out}
}

func KBSatisfiesAgentRequirements(caps types.KBCapabilities, agentMode string, allowedTools []string) bool {
	f := DeriveKBFilterForAgent(agentMode, allowedTools)
	if f.IsEmpty() { return true }
	for _, c := range f.AnyOf {
		if hasCap(caps, c) { return true } // any_of 语义：任一命中即通过
	}
	return false
}
```

能力常量：`CapVector="vector"`、`CapKeyword="keyword"`、`CapWiki="wiki"`、`CapGraph="graph"`、`CapFAQ="faq"`。工具能力表 `ToolCapabilityRequirements`：wiki 系列工具 `AllOf:{CapWiki}`，`knowledge_search`/`grep_chunks` 等 `AnyOf:{CapVector,CapKeyword}`。

**效果**：quick-answer（RAG-only）agent 在 `KBSelectionMode=all` 时过滤掉纯 wiki KB（无 vector/keyword 索引），避免推荐「agent 无法回答」的 KB。

```go
func wikiSuggestionFromPage(page *types.WikiPage, locale string) string {
	if page == nil { return "" }
	title := strings.TrimSpace(page.Title)
	if title == "" { return "" }
	switch page.PageType {
	case types.WikiPageTypeConcept:
		if isEnglishLocale(locale) { return "What is " + title + "?" }
		return "什么是" + title + "？"
	case types.WikiPageTypeEntity, types.WikiPageTypeSummary:
		if isEnglishLocale(locale) { return "Tell me about " + title }
		return "介绍一下" + title
	default:
		return title
	}
}

	case "en-US", "en", "en-GB":
		return true
	}
	return false
}
```

### 3.6 多样性算法逐字拆解（`:788-820`）

```go
// 桶内 shuffle
bucketKeys := make([]string, 0, len(buckets))
for k, qs := range buckets {
	bucketKeys = append(bucketKeys, k)
	rand.Shuffle(len(qs), func(i, j int) { qs[i], qs[j] = qs[j], qs[i] })
	buckets[k] = qs
}
// 桶间 shuffle
rand.Shuffle(len(bucketKeys), func(i, j int) {
	bucketKeys[i], bucketKeys[j] = bucketKeys[j], bucketKeys[i]
})

// round-robin：每轮每个桶取 1 条
knowledgeResult := make([]types.SuggestedQuestion, 0, limit)
offsets := make(map[string]int, len(bucketKeys))
for len(knowledgeResult) < limit {
	picked := false
	for _, key := range bucketKeys {
		if len(knowledgeResult) >= limit { break }
		qs := buckets[key]
		idx := offsets[key]
		if idx < len(qs) {
			knowledgeResult = append(knowledgeResult, qs[idx])
			offsets[key] = idx + 1
			picked = true
		}
	}
	if !picked { break }
}
```

- 桶 key：FAQ/document 用 `chunk.KnowledgeID`（一文档一桶）；wiki 用 `page.ID`。
- shuffle 用 `math/rand` 全局源（Go 1.20+ 自动播种）。
- 每轮每个桶最多贡献 1 条；终止条件是填满 `limit` 或某轮 `picked==false`（所有桶耗尽）。

#### `mergeHybridStarterSuggestions`（curated 2/3 + knowledge 1/3 + 不足互补）

```go
func mergeHybridStarterSuggestions(curated, knowledge []types.SuggestedQuestion, limit int) []types.SuggestedQuestion {
	if limit <= 0 { return []types.SuggestedQuestion{} }
	knowledgeSlots := 0
	if limit > 1 { knowledgeSlots = (limit + 1) / 3 } // 约 1/3，向上取整
	curatedSlots := limit - knowledgeSlots
	result := make([]types.SuggestedQuestion, 0, limit)
	seen := make(map[string]bool, limit)
	appendFrom := func(items []types.SuggestedQuestion, max int) {
		added := 0
		for _, item := range items {
			if len(result) == limit || (max >= 0 && added == max) { return }
			key := strings.ToLower(strings.TrimSpace(item.Question))
			if key == "" || seen[key] { continue }
			seen[key] = true
			result = append(result, item)
			added++
		}
	}
	appendFrom(curated, curatedSlots)
	appendFrom(knowledge, knowledgeSlots)
	appendFrom(curated, -1) // 不足用 curated 补位
	appendFrom(knowledge, -1)
	return result
}
```

槽位分配：`limit=6` → knowledgeSlots=2、curatedSlots=4；`limit=3` → 1/2；`limit=2` → 1/1；`limit=1` → 0/1。

### 3.9 可移植要点（Starters）

1. **能力门控**：`DeriveKBFilterForAgent`（agent mode 隐式约束 + 工具派生约束并集）+ `KBSatisfiesAgentRequirements`（any_of 语义），把「agent 能干什么」转化为「KB 是否可回答」。
2. **双 shuffle + round-robin**：分桶（knowledgeID/pageID）→ 桶内 shuffle → 桶间 shuffle → 每轮每桶取 1 条，几行代码实现「结果尽量覆盖不同文档」。
3. **按模式合并 + 不足互补**：`(limit+1)/3` 整数除法分配 knowledge 1/3 槽位，`appendFrom(..., -1)` 两次补位填满。
4. **tag scope 解析分离 ownership 与 selection**：`KnowledgeBaseIDs`（父 KB，仅 ownership 边界）、`KnowledgeIDs`（document 标签经关系表解析）、`TagIDsByTenant`（FAQ chunk 标签过滤）。
5. **跨租户 shared KB 的 chunk 归属分组**：`groupKBIDsByEffectiveTenant` 把 KB 映射到 chunk 行真正所在的源租户，无权限静默丢弃（不 403）。
6. **采集池放大**：DB 层随机取 5 倍候选（下限 20），内存再 dedup + 分桶 + round-robin，最终条数严格受 limit 控制。
7. **大小写敏感度差异（移植陷阱）**：主采集 `seen` 是精确匹配（`seen[q.Question]`），`mergeHybridStarterSuggestions` 去重键是 `strings.ToLower(strings.TrimSpace(...))`（大小写不敏感）。
8. **安全降级**：tag 解析失败/KB 列表失败/chunk 查询失败都是 `logger.ErrorWithFields` 后降级返回已有内容，而非整体失败（唯一例外是 API key 权限校验失败返回 403）。

### 4.3 `generate` 分派逻辑（逐字）

```go
func (s *messageSuggestionService) generate(ctx, message *types.Message, answer string, config types.FollowUpSuggestionConfig) (types.SuggestionItems, types.TokenUsage, error) {
	count := config.Count
	if count < 1 { count = 3 }
	generationContext, err := s.buildGenerationContext(ctx, message, config.MaxContextTurns)
	if err != nil { return nil, types.TokenUsage{}, err }
	var generated, knowledge types.SuggestionItems
	var usage types.TokenUsage
	var modelErr error
	if config.Mode == types.SuggestionModeGenerated || config.Mode == types.SuggestionModeHybrid {
		generated, usage, modelErr = s.generateWithModel(ctx, message, answer, generationContext, config, count)
	}
	needKnowledge := config.Mode == types.SuggestionModeKnowledge || config.Mode == types.SuggestionModeHybrid ||
		(modelErr != nil && config.KnowledgeFallback)
	if needKnowledge {
		knowledgeLimit := count
		if config.Mode == types.SuggestionModeGenerated {
			knowledgeLimit = count - len(generated) // generated 失败回退时补剩余坑位
		}
		knowledge, err = s.generateFromKnowledge(ctx, message, answer, generationContext, knowledgeLimit)
		if err != nil && modelErr == nil { modelErr = err }
	}
	if config.Mode == types.SuggestionModeHybrid {
		generated = mergeHybridSuggestionItems(generated, knowledge, count)
	} else {
		generated = mergeSuggestionItems(generated, knowledge, count)
	}
	if len(generated) > 0 { return generated, usage, nil }
	if modelErr != nil { return nil, usage, modelErr }
	return generated, usage, nil
}
```

分派要点：
- `count` 默认 3。
- `generated`：只走模型；失败且 `KnowledgeFallback` 时用 knowledge 补 `count - len(generated)` 坑位。
- `knowledge`：跳过模型，只走 `generateFromKnowledge`。
- `hybrid`：两者都跑，`mergeHybridSuggestionItems`（knowledge 占 1/3 坑位）。

### 4.4 `generateFromKnowledge` + `rankKnowledgeSuggestions`

```go
func (s *messageSuggestionService) generateFromKnowledge(
```

#### 相关性打分（`rankKnowledgeSuggestions` + 打分函数，逐字）

```go
func rankKnowledgeSuggestions(candidates []types.SuggestedQuestion, contextText string) {
	contextTokens := suggestionRelevanceTokens(contextText)
	contextNormalized := searchutil.NormalizeContent(contextText)
	sort.SliceStable(candidates, func(i, j int) bool {
		left := knowledgeSuggestionRelevance(candidates[i].Question, contextTokens, contextNormalized)
		right := knowledgeSuggestionRelevance(candidates[j].Question, contextTokens, contextNormalized)
		return left > right
	})
}

func knowledgeSuggestionRelevance(question string, contextTokens map[string]struct{}, contextNormalized string) float64 {
	questionNormalized := searchutil.NormalizeContent(question)
	if questionNormalized == "" { return 0 }
	questionTokens := suggestionRelevanceTokens(question)
	score := searchutil.Jaccard(questionTokens, contextTokens)
	if overlap := suggestionTokenOverlap(questionTokens, contextTokens); overlap > score {
		score = overlap
	}
	if searchutil.IsContentContained(questionNormalized, contextNormalized) {
		score++
	}
	return score
}

func suggestionRelevanceTokens(text string) map[string]struct{} {
	raw := searchutil.TokenizeSimple(text)
	cleaned := make(map[string]struct{}, len(raw))
	for token := range raw {
		token = strings.TrimFunc(token, func(r rune) bool {
			return !unicode.IsLetter(r) && !unicode.IsNumber(r)
		})
		if len([]rune(token)) > 1 { cleaned[token] = struct{}{} }
	}
	return cleaned
}

func suggestionTokenOverlap(candidate, contextTokens map[string]struct{}) float64 {
	if len(candidate) == 0 || len(contextTokens) == 0 { return 0 }
	intersection := 0
	for token := range candidate {
		if _, ok := contextTokens[token]; ok { intersection++ }
	}
	return float64(intersection) / float64(len(candidate))
}
```

打分合并（三个独立得分取较大/加和）：
1. `score := Jaccard(questionTokens, contextTokens)`（[0,1]）。
2. `overlap = |candidate∩context| / |candidate|`（overlap coefficient，非对称）；`overlap > score` 时 `score = overlap`（取较大者，非相加）。
3. `IsContentContained(questionNormalized, contextNormalized)` 为真 → `score++`（+1）。
4. `sort.SliceStable` 降序稳定排序。

底层 `searchutil`（`internal/searchutil/textutil.go`）：
- `TokenizeSimple`：含中文走 jieba `CutForSearch` 搜索模式分词，否则 `strings.Fields`；过滤单 rune 与纯标点。
- `Jaccard`：`|intersection| / |union|`，小集合驱动大集合。
- `NormalizeContent`：lower + TrimSpace + 空白压缩成单空格。
- `IsContentContained`：`strings.Contains(normalizedLong, normalizedShort)`。

### 4.5 `mergeHybridSuggestionItems`（knowledge 占 1/3 坑位，逐字）

```go
func mergeHybridSuggestionItems(model, knowledge types.SuggestionItems, limit int) types.SuggestionItems {
	if limit <= 0 { return types.SuggestionItems{} }
	knowledgeSlots := 0
	if limit > 1 { knowledgeSlots = (limit + 1) / 3 }
	modelSlots := limit - knowledgeSlots
	result := make(types.SuggestionItems, 0, limit)
	seen := make(map[string]struct{}, limit)
	appendFrom := func(items types.SuggestionItems, max int) {
		added := 0
		for _, item := range items {
			if len(result) == limit || (max >= 0 && added == max) { return }
			key := normalizeSuggestionText(item.Text)
			if key == "" { continue }
			if _, ok := seen[key]; ok { continue }
			seen[key] = struct{}{}
			result = append(result, item)
			added++
		}
	}
	appendFrom(model, modelSlots)
	appendFrom(knowledge, knowledgeSlots)
	appendFrom(model, -1)
	appendFrom(knowledge, -1)
	return result
}
```

`mergeSuggestionItems`（非 hybrid）：primary 优先填满 limit，fallback 补齐，去重键 `normalizeSuggestionText`。

### 4.6 上下文构建（`buildGenerationContext` / `buildSuggestionGenerationContext`）

要点：
- `maxTurns < 1` 时默认 2；取最近 `maxTurns*4+8` 条消息（放宽取数，避免不完整轮次挤掉完整 user/assistant 对）。
- 用 `groupSuggestionConversationTurns` 把消息聚合成 user/assistant 轮次。
- `currentQuery` 用 `turn.user.Content`（**用户可见 query**，而非 RenderedContent——后者含完整 RAG prompt 与原始 chunk，故意排除）。
- 历史取 `maxTurns-1` 个**已完成**轮次（当前轮计入 maxTurns）。
- 证据 `buildSuggestionEvidence`：按引用 Score 降序、去重、最多 5 条，每条 `[N] title: snippet`（snippet 截 500 rune 后压空白），同时返回去重后的 KnowledgeID 列表作 `ActualKnowledgeIDs`。

### 4.7 辅助函数逐字

```go
func answerEndsWithQuestion(answer string) bool {
	answer = strings.TrimSpace(trailingCitationTags.ReplaceAllString(answer, ""))
	return strings.HasSuffix(answer, "?") || strings.HasSuffix(answer, "？")
}

func emptySuggestionSection(value string) string {
	if value = strings.TrimSpace(value); value != "" { return value }
	return "(none)"
}

func truncateRunes(value string, limit int) string {
	runes := []rune(value)
	if len(runes) <= limit { return value }
	return string(runes[:limit])
}

func normalizeSuggestionText(value string) string {
	return strings.Map(func(r rune) rune {
		if unicode.IsSpace(r) || strings.ContainsRune("?？!！,，.。:：;；\"'", r) {
			return -1
		}
		return unicode.ToLower(r)
	}, strings.TrimSpace(value))
}

func suggestionErrorCode(err error) string {
	if err == nil { return "" }
	if errors.Is(err, gorm.ErrRecordNotFound) { return "not_found" }
	value := strings.ToLower(err.Error())
	switch {
	case strings.Contains(value, "model"): return "model_error"
	case strings.Contains(value, "json"):   return "invalid_model_output"
	default:                                return "generation_error"
	}
}
```

### 4.8 可移植要点（FollowUps）

1. **接地约束（grounding）**：system prompt 强制问题落在当前 question/answer/evidence 边界内、检索型问题自带实体名、最多 1/3 探索相邻、证据只当数据不当指令——这是质量与安全核心。
2. **相关性重排纯规则零 LLM 成本**：Jaccard + overlap coefficient + 包含关系 +1，三路打分取较大/加和，稳定排序 top-N。
3. **抑制前置**：所有低成本判定（未启用/fallback/空答案/反问）都在模型调用前短路并落库 `suppressed` + `suppression_reason`。
4. **租约去重**：`AcquireGeneration` 串行化并发生成，ready/suppressed 命中复用，regenerate 才重新抢占。
5. **租户隔离**：模型与知识召回都按 `AgentTenantID` 切换 context；受限 API key 禁用「按实际证据 ID 召回」优化。
6. **可观测字段**：PromptTokens/CompletionTokens/LatencyMs/ErrorCode/GeneratedAt 全落库，错误码归一为 `not_found`/`model_error`/`invalid_model_output`/`generation_error`。

### 6.4 归属校验 `ValidateAttribution`（逐字，`message_suggestion.go:219-254`）

```go
func (s *messageSuggestionService) ValidateAttribution(ctx, sessionID, query string, attribution *types.SuggestionAttribution) error {
	if attribution == nil { return nil }
	if strings.TrimSpace(attribution.SuggestionSetID) == "" || strings.TrimSpace(attribution.QuestionID) == "" {
		return errors.New("invalid suggestion attribution")
	}
	set, err := s.repo.GetByID(ctx, types.MustTenantIDFromContext(ctx), sessionID, attribution.SuggestionSetID)
	if err != nil { return err }
	if set.Status != types.SuggestionStatusReady {
		return errors.New("invalid suggestion attribution")
	}
	found := false
	for _, question := range set.Questions {
		if question.ID == attribution.QuestionID && strings.TrimSpace(question.Text) == strings.TrimSpace(query) {
			found = true
			break
		}
	}
	if !found { return errors.New("invalid suggestion attribution") }
	return nil
}
```

逻辑：`attribution == nil` 放行；ID 空返回无效；`GetByID` 取 set（取不到原样返回 error）；set 必须 `ready`；要求存在 item 同时满足 `ID == attribution.QuestionID` **且** `Text == query`（点击的问题文本必须与用户实际发出的 query 逐字一致，防伪造）。

### 6.6 可移植要点（接口/触发）

1. **异步生成 + `context.WithoutCancel`**：同步置完成，goroutine 摘出后台 context 生成，失败仅 warn。
2. **attribution 前置校验（anti-spoof）**：发起 SSE 前校验 set ready + question 文本与 query 逐字一致 + ID 真实属于该 set，杜绝伪造埋点。
3. **引用式埋点**：事件表只存 `suggestion_set_id` + `question_id` + `event_type`，不复制文本。
4. **status 驱动三段式协议**：`generating`（202 轮询）/`ready`/`suppressed`（含 reason）/`failed`（含 error_code），错误不抛给调用方而是落成带状态 set。
5. **`regenerate` 语义内聚服务端**：仅 `Enabled && AllowRegenerate` 才允许，成功后再内部写 regenerate 事件（前端不被允许上报）。
6. **错误映射用字符串匹配**：移植时若要更稳健可改显式 sentinel error 类型。

## 8. 移植路线图

### 第 0 步：身份模型与缓存键（必须最先做）

一张 `message_suggestion_sets` 表，缓存键 = `(tenant_id, assistant_message_id, placement, config_hash, locale)` 唯一索引。**别用随机 ID 当缓存键**，用这五元组。

### 第 1 步：lease 抢占（并发去重的核心，性价比最高）

`AcquireGeneration` 的三段式：`ON CONFLICT DO NOTHING` 抢占 → 读现有行判断状态 → 乐观 `UPDATE` 二次抢占。这一步做好，前端重连/多端的重复生成问题直接消失。

### 第 2 步：抑制规则前置

把「未启用 / fallback 答案 / 空答案 / 反问」四个低成本判定放在模型调用之前，`suppress(reason)` 落库。这是「不给用户推废话」+ 可观测的基础。

### 第 3 步：知识来源采集（存量资产优先）

先只做 `faq`（StandardQuestion）+ `document`（GeneratedQuestions 第一条）+ `wiki`（标题套模板）三种存量来源 + `agent_config`（手写精选），不做 LLM 现场生成。加「分桶 + 双 shuffle + round-robin」保证多样性。这一步已经能交付「开场推荐」。

### 第 4 步：追问 LLM 生成（generated 模式）

照抄 §5 的 system prompt（接地约束是质量核心）和 user prompt 四段拼装。temperature=0.3、max_tokens=700、关 thinking。

### 第 5 步：相关性重排（knowledge 模式）

纯规则：Jaccard + overlap coefficient + 包含关系 +1，零 LLM 成本排序存量问题，作为 generated 失败的回退。

### 第 6 步：hybrid 合并

`(limit+1)/3` 分配 knowledge 1/3 坑位，`appendFrom(..., -1)` 不足互补。

### 第 7 步：埋点与归属（产品分析）

events 表（存 ID 不存文本）+ attribution 前置校验（点击问题文本必须与用户实际 query 逐字一致）。

### 移植时必须一起带上的硬约束

1. **缓存键唯一索引**含 `config_hash`，配置变化自动失效缓存。
2. **lease 抢占三段式**，`ready`/`suppressed` 命中即复用，`regenerate` 才重新抢占。
3. **`SuggestionItems` 自定义 Valuer/Scanner**，nil 归一为 `[]`。
4. **JSON 字段按数据库方言分支**（`db.Name()` 判断），移植到单库只保留对应方言。
5. **能力门控**（quick-answer 不采 wiki；KB 能力过滤用 any_of 并集）。
6. **抑制 reason / 错误码统一枚举**，可观测性靠这两个字段。
7. **attribution 前置校验**（anti-spoof），点击文本必须与 query 逐字一致。
8. **异步生成用 `context.WithoutCancel`**，失败仅 warn 不回滚主流程。
